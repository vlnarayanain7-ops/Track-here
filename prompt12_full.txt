  );
}

// src/lib/statsBuckets.js: minutesByDay is a Map('YYYY-MM-DD' -> minutes) from the SAME endpoint the heatmap uses
const key = d => new Intl.DateTimeFormat('en-CA').format(d);
const addDays = (d, n) => { const x = new Date(d); x.setDate(x.getDate() + n); return x; };
export function buckets(map, range, weekStartsOn = 1, now = new Date()) {
  const today = new Date(now); today.setHours(12, 0, 0, 0);
  const val = d => (d > today ? 0 : map.get(key(d)) ?? 0);
  if (range === 'week') {
    const start = addDays(today, -((today.getDay() - weekStartsOn + 7) % 7));
    return Array.from({ length: 7 }, (_, i) => { const d = addDays(start, i);
      return { label: d.toLocaleDateString('en', { weekday: 'narrow' }), title: d.toDateString(), minutes: val(d) }; });
  }
  if (range === 'month') {
    const y = today.getFullYear(), m = today.getMonth(), n = new Date(y, m + 1, 0).getDate();
    return Array.from({ length: n }, (_, i) => { const d = new Date(y, m, i + 1, 12);
      return { label: String(i + 1), title: d.toDateString(), minutes: val(d) }; });
  }
  const y = today.getFullYear();                                       // year: one bar per month
  return Array.from({ length: 12 }, (_, m) => {
    let total = 0; for (let d = 1; d <= new Date(y, m + 1, 0).getDate(); d++) total += val(new Date(y, m, d, 12));
    const first = new Date(y, m, 1, 12);
    return { label: first.toLocaleString('en', { month: 'narrow' }), title: first.toLocaleString('en', { month: 'long' }), minutes: total };
  });
}

// src/components/StatsPanel.jsx
import { useMemo, useState } from 'react';
import { buckets } from '../lib/statsBuckets';
import { fmtDuration } from '../lib/format';
export function StatsPanel({ minutesByDay, bySubject = [], weekStartsOn = 1, onRangeChange }) {
  const [range, setRange] = useState('week');
  const rows = useMemo(() => buckets(minutesByDay, range, weekStartsOn), [minutesByDay, range, weekStartsOn]);
  const total = rows.reduce((a, r) => a + r.minutes, 0), max = Math.max(...rows.map(r => r.minutes), 1);
  const pick = r => { setRange(r); onRangeChange?.(r); };            // refetch bySubject with ?range=
  return (
    <div className="stack">
      <div className="hd">
        <div><div className="lab">STUDIED</div><div className="bignum">{fmtDuration(total)}</div></div>
        <div className="seg" role="group" aria-label="Range">
          {['week', 'month', 'year'].map(r => <button key={r} aria-pressed={range === r} onClick={() => pick(r)}>{r[0].toUpperCase() + r.slice(1)}</button>)}
        </div>
      </div>
      {total === 0
        ? <p className="note">No study time this {range} yet. Start a focus session and it will show up here.</p>
        : <div className="card"><div className="bars" role="img" aria-label={`Studied time this ${range}`}>
            {rows.map((r, i) => (
              <div key={i}><i title={`${r.title}: ${fmtDuration(r.minutes)}`} style={{ height: Math.max((r.minutes / max) * 140, r.minutes ? 6 : 2) }} />
                <span>{range === 'month' && i % 5 && i !== 0 ? '' : r.label}</span></div>))}
          </div></div>}
      {bySubject.length > 0 && (
        <div><h2 style={{ fontSize: 16, margin: '8px 0' }}>By subject</h2>
          {bySubject.map(s => (
            <div key={s.id} style={{ padding: '10px 0', borderTop: '1px solid var(--line)' }}>
              <div style={{ display: 'flex', justifyContent: 'space-between' }}><span>{s.name}</span><span className="note">{fmtDuration(s.minutes)}</span></div>
              <div style={{ height: 4, borderRadius: 2, background: 'var(--line)', marginTop: 6 }}>
                <i style={{ display: 'block', height: 4, borderRadius: 2, background: `var(--c-${s.color || 'slate'})`, width: `${(s.minutes / Math.max(...bySubject.map(x => x.minutes), 1)) * 100}%` }} /></div>
            </div>))}
        </div>)}
    </div>
  );
}

const ymd = d => new Intl.DateTimeFormat('en-CA').format(d);
const level = m => (m <= 0 ? 0 : m < 25 ? 1 : m < 60 ? 2 : m < 120 ? 3 : m < 240 ? 4 : 5);        // 25+ min = streak day
const fmt = m => (m >= 60 ? `${Math.floor(m / 60)}h ${m % 60}m` : `${m}m`);

export function buildMonths(minutesByDay, weekStartsOn = 1, count = 12, now = new Date()) {
  const today = new Date(now); today.setHours(12, 0, 0, 0);
  const months = [];
  for (let k = count - 1; k >= 0; k--) {
    const first = new Date(today.getFullYear(), today.getMonth() - k, 1, 12);
    const days = new Date(first.getFullYear(), first.getMonth() + 1, 0).getDate();
    const offset = (first.getDay() - weekStartsOn + 7) % 7;                 // blank cells before day 1
    const cells = [];
    for (let d = 1; d <= days; d++) {
      const date = new Date(first.getFullYear(), first.getMonth(), d, 12), idx = offset + d - 1;
      if (date > today) continue;                                           // future days stay empty
      cells.push({ key: ymd(date), date, col: Math.floor(idx / 7), row: idx % 7, minutes: minutesByDay.get(ymd(date)) ?? 0 });
    }
    months.push({ id: ymd(first), label: first.toLocaleString('en', { month: 'short' }), cols: Math.floor((offset + days - 1) / 7) + 1, cells });
  }
  return months;
}

export function MonthHeatmap({ minutesByDay, weekStartsOn = 1 }) {
  const months = useMemo(() => buildMonths(minutesByDay, weekStartsOn), [minutesByDay, weekStartsOn]);
  const scroller = useRef(null), todayKey = ymd(new Date());
  useLayoutEffect(() => { const el = scroller.current; if (el) el.scrollLeft = el.scrollWidth; }, [months.length]);
  const tip = c => `${c.date.toLocaleDateString('en-US', { weekday: 'long', month: 'short', day: 'numeric', year: 'numeric' })}${c.minutes ? ` · ${fmt(c.minutes)}` : ''}`;
  const dayLabels = ['Mon', '', 'Wed', '', 'Fri', '', ''];                  // rotate this list if weekStartsOn is not Monday
  return (
    <div>
      <div className="mh" role="img" aria-label="Study activity for the last 12 months">
        <div className="mh-days" aria-hidden="true">{dayLabels.map((d, i) => <span key={i}>{d}</span>)}</div>
        <div className="mh-scroll" ref={scroller}>
          <div className="mh-row">
            {months.map(m => (
              <div key={m.id}>
                <div className="mh-grid" style={{ gridTemplateColumns: `repeat(${m.cols}, var(--cell))`, gridTemplateRows: 'repeat(7, var(--cell))' }}>
                  {m.cells.map(c => (
                    <i key={c.key} className={`mh-cell${c.key === todayKey ? ' today' : ''}`} title={tip(c)}
                      style={{ gridColumn: c.col + 1, gridRow: c.row + 1, background: `var(--heat-${level(c.minutes)})` }} />))}
                </div>
                <div className="mh-label">{m.label}</div>
              </div>))}
          </div>
        </div>
      </div>
      <div className="leg">Less {[0, 1, 2, 3, 4, 5].map(l => <i key={l} style={{ background: `var(--heat-${l})` }} />)} More</div>
    </div>
  );
}
