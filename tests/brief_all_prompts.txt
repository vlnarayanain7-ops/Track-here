import {useEffect,useId,useRef,type ButtonHTMLAttributes,type InputHTMLAttributes,type ReactNode} from 'react';

export type ColorKey='blue'|'green'|'amber'|'rose'|'violet'|'cyan'|'orange'|'slate';

export function Button({variant='secondary',size='md',loading,icon,children,...p}:
  ButtonHTMLAttributes<HTMLButtonElement>&{variant?:'primary'|'secondary'|'ghost'|'danger';size?:'md'|'lg';loading?:boolean;icon?:ReactNode}){
  const cls=['btn',variant==='primary'&&'btn-primary',variant==='ghost'&&'btn-ghost',variant==='danger'&&'btn-danger',size==='lg'&&'btn-lg'].filter(Boolean).join(' ');
  return <button {...p} className={cls} disabled={p.disabled||loading} aria-busy={loading}>{icon}{loading?'Working…':children}</button>;
}
export const Card=({children,style}:{children:ReactNode;style?:React.CSSProperties})=><section className="card" style={style}>{children}</section>;

export function Field({label,hint,error,...p}:InputHTMLAttributes<HTMLInputElement>&{label:string;hint?:string;error?:string}){
  const id=useId();
  return <label className="field" htmlFor={id} data-invalid={!!error}>{label}
    <input id={id} aria-invalid={!!error} aria-describedby={`${id}-d`} {...p}/>
    {(error||hint)&&<small id={`${id}-d`} className={error?'err':''} role={error?'alert':undefined}>{error||hint}</small>}</label>;
}
export const Badge=({color='slate',children}:{color?:ColorKey;children:ReactNode})=>
  <span className="badge" style={{color:`var(--c-${color})`,background:`var(--c-${color}-bg)`}}>{children}</span>;

export const EmptyState=({icon,title,text,action}:{icon?:ReactNode;title:string;text?:string;action?:ReactNode})=>
  <div className="empty">{icon}<h3>{title}</h3>{text&&<p style={{margin:0,maxWidth:420}}>{text}</p>}{action}</div>;

export const Skeleton=({w='100%',h=14}:{w?:string|number;h?:number})=><div className="skeleton" style={{width:w,height:h}} aria-hidden/>;

export function Modal({open,onClose,title,children}:{open:boolean;onClose:()=>void;title:string;children:ReactNode}){
  const ref=useRef<HTMLDialogElement>(null);
  useEffect(()=>{const d=ref.current;if(!d)return;open&&!d.open?d.showModal():!open&&d.open&&d.close()},[open]);
  return <dialog ref={ref} className="modal" onClose={onClose} aria-labelledby="m-t">

      <h2 id="m-t" style={{fontSize:20}}>{title}</h2><button className="btn btn-ghost" aria-label="Close" onClick={onClose}>✕</button></div>{children}</dialog>;

// server/streak.js: current streak in the user's own timezone; 25+ focus minutes in a local day counts
export function currentStreak(sessions,tz,graceDays=0,now=new Date()){
  const fmt=new Intl.DateTimeFormat('en-CA',{timeZone:tz}), mins=new Map();
  for(const s of sessions){const d=fmt.format(new Date(s.startedAt)); mins.set(d,(mins.get(d)||0)+s.durationSec/60)}
  const ok=d=>(mins.get(d)||0)>=25;
  const day=new Date(now);
  if(!ok(fmt.format(day))) day.setDate(day.getDate()-1);      // today is still in progress, do not break the streak yet
  let streak=0,gap=0;
  for(let i=0;i<3650;i++){
    if(ok(fmt.format(day))){streak++;gap=0} else if(++gap>graceDays) break;
    day.setDate(day.getDate()-1);
  }
  return streak;
}

async function request<T>(method:string,path:string,body?:unknown,retry=method==='GET'):Promise<T>{
  const ctrl=new AbortController(); const slow=setTimeout(()=>onWaking(true),4000); const kill=setTimeout(()=>ctrl.abort(),75000);
  try{
    const res=await fetch(BASE+path,{method,credentials:'include',signal:ctrl.signal,
      headers:{'Content-Type':'application/json',...authHeader()},body:body===undefined?undefined:JSON.stringify(body)});
    if(res.status===401){window.dispatchEvent(new Event('th:signed-out'));throw new ApiError(401,'unauthorized','Please sign in again.')}
    const data=res.status===204?null:await res.json().catch(()=>null);
    if(!res.ok) throw new ApiError(res.status,data?.error??'error',data?.message??'Something went wrong. Please try again.');
    return data as T;
  }catch(e){
    if(retry&&!(e instanceof ApiError)) return request<T>(method,path,body,false);   // one retry for network errors on reads
    throw e;
  }finally{clearTimeout(slow);clearTimeout(kill);onWaking(false)}
}
